## 一、账号存储升级（任务 24）

### 方案：SQLite → PostgreSQL

**阶段 1：SQLite（当前过渡方案）**
- 文件：api-gateway/authdata/users.json → users.db
- 优点：零依赖、快速迁移
- 缺点：单文件、并发性能差

**阶段 2：PostgreSQL（生产推荐）**
```sql
-- 用户表
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(64) UNIQUE NOT NULL,
    password_hash VARCHAR(256) NOT NULL,
    salt VARCHAR(64) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_login_at TIMESTAMP,
    is_active BOOLEAN DEFAULT TRUE
);

-- 审计日志表
CREATE TABLE audit_logs (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES users(id),
    action VARCHAR(64) NOT NULL,
    ip_address VARCHAR(45),
    user_agent TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 索引
CREATE INDEX idx_users_username ON users(username);
CREATE INDEX idx_audit_logs_user_id ON audit_logs(user_id);
CREATE INDEX idx_audit_logs_created_at ON audit_logs(created_at);
```

### 迁移脚本
```python
# api-gateway/migrate_users.py
import json
import sqlite3
from pathlib import Path

def migrate_json_to_sqlite(json_path: str, db_path: str):
    """将 users.json 迁移到 SQLite"""
    with open(json_path) as f:
        users = json.load(f)
    
    conn = sqlite3.connect(db_path)
    cursor = conn.cursor()
    
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS users (
            username TEXT PRIMARY KEY,
            password_hash TEXT NOT NULL,
            salt TEXT NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    ''')
    
    for username, data in users.items():
        cursor.execute(
            'INSERT OR REPLACE INTO users (username, password_hash, salt) VALUES (?, ?, ?)',
            (username, data['password_hash'], data['salt'])
        )
    
    conn.commit()
    conn.close()
    print(f"迁移完成：{len(users)} 个用户")
```

## 二、会话与角色持久化（任务 25）

### Redis 持久化方案

**当前问题**：会话与角色状态驻留内存，重启即失

**解决方案**：
```python
# ai-service/storage/redis_storage.py
import redis
import json
from typing import Optional

class RedisStorage:
    def __init__(self, host="localhost", port=6379, db=0):
        self.client = redis.Redis(
            host=host, port=port, db=db,
            decode_responses=True,
            socket_connect_timeout=5,
            socket_timeout=5,
        )
    
    # ===== 会话持久化 =====
    def set_session(self, session_id: str, data: dict, expire: int = 86400):
        """设置会话数据，默认 24 小时过期"""
        key = f"session:{session_id}"
        self.client.setex(key, expire, json.dumps(data))
    
    def get_session(self, session_id: str) -> Optional[dict]:
        key = f"session:{session_id}"
        data = self.client.get(key)
        return json.loads(data) if data else None
    
    def delete_session(self, session_id: str):
        self.client.delete(f"session:{session_id}")
    
    # ===== 角色持久化 =====
    def set_character(self, character_id: str, data: dict):
        key = f"character:{character_id}"
        self.client.set(key, json.dumps(data))
    
    def get_character(self, character_id: str) -> Optional[dict]:
        key = f"character:{character_id}"
        data = self.client.get(key)
        return json.loads(data) if data else None
    
    def list_characters(self) -> list:
        keys = self.client.keys("character:*")
        return [self.get_character(k.split(":")[1]) for k in keys]
    
    # ===== 对话历史 =====
    def append_dialogue(self, session_id: str, message: dict, max_length: int = 100):
        key = f"dialogue:{session_id}"
        self.client.lpush(key, json.dumps(message))
        self.client.ltrim(key, 0, max_length - 1)
    
    def get_dialogue_history(self, session_id: str, limit: int = 10) -> list:
        key = f"dialogue:{session_id}"
        messages = self.client.lrange(key, 0, limit - 1)
        return [json.loads(m) for m in messages]
```

## 三、缓存与限流外置（任务 26）

### 多实例限流共享存储

**问题**：多实例部署时限流计数独立，导致限流失效

**解决方案**：Redis 计数器
```python
# api-gateway/rate_limit_redis.py
import redis
import time

class RedisRateLimiter:
    def __init__(self, redis_client):
        self.redis = redis_client
    
    def is_allowed(self, key: str, limit: int, window: int = 60) -> bool:
        """
        滑动窗口限流
        key: 限流键（如 ip:login）
        limit: 窗口内最大请求数
        window: 窗口大小（秒）
        """
        now = time.time()
        window_start = now - window
        
        # 清理过期记录
        self.redis.zremrangebyscore(key, 0, window_start)
        
        # 获取当前窗口计数
        current = self.redis.zcard(key)
        
        if current >= limit:
            return False
        
        # 添加新记录
        self.redis.zadd(key, {str(now): now})
        self.redis.expire(key, window)
        
        return True
```

import os
import time
from pathlib import Path

def cleanup_old_files(directory: str, max_age_days: int = 7):
    """清理指定天数前的文件"""
    now = time.time()
    max_age = max_age_days * 86400
    
    for file_path in Path(directory).glob("**/*"):
        if file_path.is_file():
            file_age = now - file_path.stat().st_mtime
            if file_age > max_age:
                file_path.unlink()
                print(f"已删除：{file_path}")
